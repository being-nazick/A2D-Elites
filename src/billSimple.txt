/*
 * IMPORTANT:
 * 1. PDF size = A4 Portrait
 * 2. QR code = removed
 * 3. Signature = required
 * 4. html2pdf = local npm package
 * 5. Invoice data is HTML-escaped
 * 6. PDF generation and sharing are handled separately
 * 7. Print fallback is also A4
 */

/* HELPERS                                                                    */
/* -------------------------------------------------------------------------- */

/**
 * Safely escape values before inserting them into HTML.
 */
function escapeHtml(value) {
  return String(value ?? "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#039;");
}

/**
 * Convert a value into a valid number.
 */
function toNumber(value, fallback = 0) {
  const number = Number(value);

  return Number.isFinite(number) ? number : fallback;
}

/**
 * Make a safe filename.
 */
function safeFilename(value) {
  return String(value || "invoice")
    .replace(/[<>:"/\\|?*\x00-\x1F]/g, "_")
    .replace(/\s+/g, "_")
    .trim();
}

/**
 * Normalize invoice items.
 */
function normalizeBillItems(items) {
  if (!Array.isArray(items)) {
    return [];
  }

  return items.map((item) => {
    const qty = Math.max(0, toNumber(item?.qty));

    const price = Math.max(0, toNumber(item?.price));

    return {
      ...item,

      qty,

      price,

      amount: qty * price,

      productName: String(item?.productName || item?.name || "Item"),

      unit: String(item?.unit || "Unit"),
    };
  });
}

/**
 * Calculate total from line items.
 */
function calculateBillTotal(items) {
  return items.reduce((sum, item) => sum + item.amount, 0);
}

/**
 * Get the configured signature image.
 *
 * Recommended:
 *     data:image/png;base64,...
 *
 * A local/base64 image is much more reliable than
 * an external image URL when html2canvas creates a PDF.
 */
function getInvoiceSignature() {
  let signature = "";

  try {
    if (typeof signatureImg !== "undefined" && signatureImg) {
      signature = signatureImg;
    }
  } catch (_) {
    // signatureImg may not exist.
  }

  try {
    if (!signature && typeof SIGNATURE_IMG !== "undefined" && signatureImg) {
      signature = signatureImg;
    }
  } catch (_) {
    // SIGNATURE_IMG may not exist.
  }

  return String(signature || "").trim();
}

/**
 * Wait for all invoice images to load.
 *
 * This is especially important for the signature.
 */
async function waitForInvoiceImages(container) {
  const images = Array.from(container.querySelectorAll("img"));

  if (!images.length) {
    return;
  }

  await Promise.all(
    images.map((img) => {
      if (img.complete && img.naturalWidth > 0) {
        return Promise.resolve();
      }

      return new Promise((resolve, reject) => {
        img.onload = () => resolve();

        img.onerror = () => {
          reject(
            new Error(
              `Failed to load invoice image: ${img.alt || "unknown image"}`,
            ),
          );
        };
      });
    }),
  );
}

/**
 * Wait until the browser has had a chance to finish layout/fonts.
 */
async function waitForInvoiceRender() {
  await new Promise((resolve) => {
    requestAnimationFrame(() => {
      requestAnimationFrame(resolve);
    });
  });

  /*
   * Give web fonts a chance to finish loading.
   */
  if (document.fonts && document.fonts.ready) {
    try {
      await document.fonts.ready;
    } catch (_) {
      // Font loading failure should not stop PDF generation.
    }
  }
}

function buildBillMarkup(order) {
  if (!order || typeof order !== "object") {
    throw new Error("Invalid order data.");
  }

  /* ------------------------------ ITEMS ------------------------------ */

  const billItems = normalizeBillItems(order.items);

  if (!billItems.length) {
    throw new Error("Cannot generate bill because this order has no items.");
  }

  /* ---------------------------- SIGNATURE ---------------------------- */

  const signature = getInvoiceSignature();

  if (!signature) {
    throw new Error(
      "Authorized signature is missing. Please configure the signature before generating the bill.",
    );
  }

  /* ------------------------------ BUSINESS ------------------------------ */

  const businessName = escapeHtml(
    typeof BUSINESS_NAME !== "undefined" ? BUSINESS_NAME : "A2D'Elites",
  );

  const businessTagline = escapeHtml(
    typeof BUSINESS_TAGLINE !== "undefined" ? BUSINESS_TAGLINE : "SINCE 2025",
  );

  /* ------------------------------ CUSTOMER ------------------------------ */

  const customerName = escapeHtml(order.customerName || "—");

  const phone = escapeHtml(order.phone || "");

  const address = escapeHtml(order.address || "");

  const notes = escapeHtml(order.notes || "");

  /* ------------------------------ DATE ------------------------------ */

  const orderDate =
    order.orderDate ||
    (typeof todayStr === "function"
      ? todayStr()
      : new Date().toISOString().slice(0, 10));

  const formattedDate =
    typeof fmtDate === "function"
      ? escapeHtml(fmtDate(orderDate))
      : escapeHtml(orderDate);

  /* ------------------------------ TOTAL ------------------------------ */

  const calculatedTotal = calculateBillTotal(billItems);

  const storedTotal = toNumber(order.total, NaN);

  /*
   * If order.total is a valid number,
   * preserve it.
   *
   * Otherwise calculate from items.
   */
  const total = Number.isFinite(storedTotal)
    ? Math.max(0, storedTotal)
    : calculatedTotal;

  /* ---------------------------- PAYMENT ---------------------------- */

  const amountPaid = Math.max(0, toNumber(order.amountPaid));

  const balanceDue = Math.max(0, total - amountPaid);

  const excessPaid = Math.max(0, amountPaid - total);

  /* ------------------------------ STATUS ------------------------------ */

  const orderStatus = escapeHtml(order.orderStatus || "Pending");

  const paymentStatus = escapeHtml(
    order.paymentStatus || (balanceDue <= 0 ? "Paid" : "Pending"),
  );

  /* -------------------------- INVOICE NUMBER -------------------------- */

    order.invoiceNo ||
    `INV-${String(order.id || "invoice")
      .toUpperCase()

            </td>

              <div class="item-name">
                ${escapeHtml(item.productName)}
              </div>

              <div class="item-unit">
                ${escapeHtml(item.unit)}
              </div>

            </td>

            <td class="cell rate-cell">
              ${fmtINR(item.price)}
            </td>

            <td class="cell qty-cell">
              ${item.qty}
            </td>

            <td class="cell amount-cell">
              ${fmtINR(item.amount)}
            </td>

          </tr>
        `,
    )
    .join("");

  if (amountPaid > 0) {
    paymentHtml += `
      <div class="payment-row">

        <span>
          Amount Paid
        </span>

        <strong>
          ${fmtINR(amountPaid)}
        </strong>

      </div>
    `;
  }

  if (balanceDue > 0) {
    paymentHtml += `
      <div class="payment-row balance-row">

        <span>
          Balance Due
        </span>

        <strong>
          ${fmtINR(balanceDue)}
        </strong>

      </div>
    `;
  } else {
    paymentHtml += `
      <div class="payment-row paid-row">

        <span>
          Payment Status
        </span>

        <strong>
          PAID
        </strong>

      </div>
    `;
  }

  if (excessPaid > 0) {
    paymentHtml += `
      <div class="payment-row">

        </span>

          </div>

          <div class="notes-text">

/* -------------------------------------------------------------------------- */
/* GENERATE BILL                                                             */
/* -------------------------------------------------------------------------- */

async function generateBill(order) {
  let el = null;

  try {
    /* --------------------------- VALIDATION --------------------------- */

    if (!order || typeof order !== "object") {
      throw new Error("Invalid order. Cannot generate bill.");
    }

    if (!Array.isArray(order.items) || order.items.length === 0) {
      throw new Error("Cannot generate bill because this order has no items.");
    }

    /* -------------------------- BUILD HTML -------------------------- */

    const billMarkup = buildBillMarkup(order);

    /* ---------------------- CREATE RENDER AREA ---------------------- */

    el = document.createElement("div");

    /*
     * IMPORTANT:
     *
     * Do not use left:-9999px.
     *
     * html2canvas can produce incorrect positioning
     * for elements far outside the viewport.
     *
     * Instead keep it renderable but invisible.
     */
    el.style.position = "relative";
    el.style.width = "703px";
    el.style.minHeight = "1123px";
    el.style.margin = "0";
    el.style.padding = "0";
    el.style.background = "#ffffff";
    el.style.opacity = "1";
    el.style.visibility = "visible";
    el.style.pointerEvents = "none";

    el.innerHTML = billMarkup;

    document.body.appendChild(el);

    console.log("INVOICE ELEMENT:", el);
    console.log("INVOICE HEIGHT:", el.scrollHeight);
    console.log("INVOICE WIDTH:", el.scrollWidth);
    console.log("INVOICE HTML LENGTH:", el.innerHTML.length);
    /* ---------------------- WAIT FOR RENDER ---------------------- */

    await waitForInvoiceRender();

    await waitForInvoiceImages(el);

    /* ---------------------- VERIFY SIGNATURE ---------------------- */

    const signatureElement = el.querySelector(".signature-image");

    if (
      !signatureElement ||
      !signatureElement.complete ||
      signatureElement.naturalWidth <= 0
    ) {
      throw new Error(
        "Authorized signature could not be loaded. Bill generation stopped.",
      );
    }

    /* ------------------------- FILENAME ------------------------- */

    const invoiceNumber =
      order.invoiceNo ||
      `INV-${String(order.id || "invoice")
        .toUpperCase()
        .slice(-6)}`;

    const filename = `${safeFilename(`bill-${invoiceNumber}`)}.pdf`;

    /* ------------------------- PDF OPTIONS ------------------------- */

    const options = {
      /*
       * 8mm around the A4 page.
       */
      margin: [8, 8, 8, 8],

      filename,

      image: {
        type: "jpeg",

        quality: 0.98,
      },

      html2canvas: {
        scale: 2,
        useCORS: true,
        allowTaint: false,
        logging: true,
        backgroundColor: "#ffffff",
        imageTimeout: 15000,
        scrollX: 0,
        scrollY: 0,
        windowWidth: 794,
        windowHeight: 1123,
      },

      jsPDF: {
        unit: "mm",

        /*
         * IMPORTANT:
         * A4 instead of A5.
         */
        format: "a4",

        orientation: "portrait",

        compress: true,
      },

      pagebreak: {
        mode: ["css", "legacy"],

        avoid: [
          "tr",
          "img",
          ".no-break",
          ".signature-section",
          ".signature-box",
          ".total-box",
          ".payment-section",
        ],
      },
    };
    /* ---------------------- GENERATE PDF ONCE ---------------------- */

    const pdfDataUri = await html2pdf()
      .set(options)
      .from(el)
      .outputPdf("datauristring");

    const commaIndex = pdfDataUri.indexOf(",");

    if (commaIndex === -1) {
      throw new Error("Unable to generate PDF data.");
    }

    const base64Data = pdfDataUri.slice(commaIndex + 1);

    /* ---------------------- ANDROID / BROWSER ---------------------- */

    const isCapacitor =
      typeof window !== "undefined" && window.Capacitor?.isNativePlatform?.();

    if (isCapacitor) {
      /* ------------------------ ANDROID ------------------------ */

      const file = await Filesystem.writeFile({
        path: filename,
        data: base64Data,
        directory: Directory.Cache,
        recursive: true,
      });

      try {
        await Share.share({
          title: "Dairy Bill",
          text: `Bill for ${order.customerName || "customer"}`,
          url: file.uri,
          dialogTitle: "Share or print bill",
        });
      } catch (shareError) {
        /*
         * User may have pressed Back / closed the share sheet.
         * Do not generate another bill or open the print fallback.
         */
        console.warn("Bill sharing cancelled or failed:", shareError);
      }
    } else {
      /* ------------------------ LAPTOP / DESKTOP ------------------------ */

      const link = document.createElement("a");

      link.href = pdfDataUri;

      link.download = filename;

      document.body.appendChild(link);

      link.click();

      link.remove();
    }

    /* ------------------------ CLEAN UP ------------------------ */

    if (el && el.parentNode) {
      el.parentNode.removeChild(el);
      el = null;
    }
  } catch (error) {
    /* ------------------------ FINAL ERROR ------------------------ */

    console.error("Bill generation failed:", error);

    if (el && el.parentNode) {
      el.parentNode.removeChild(el);
      el = null;
    }

    alert(error?.message || "Unable to generate the bill. Please try again.");
  }
}
